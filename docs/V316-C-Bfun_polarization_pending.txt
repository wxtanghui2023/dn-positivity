/-- **③′ 积分极化**：`B(u,u) − B(v,v) = B(v,h) + B(h,v) + B(h,h)`（`h = u − v`）。
只用点态恒等式（`ring`）＋ `integral_sub`／`integral_add`；
可积性作**显式前提**（与 `kernel_bound` 既有 7 条可积性假设的风格一致，不造黑箱）。 -/
theorem Bfun_polarization {u v : ℝ → ℝ}
    (hUU : Integrable (fun p : ℝ × ℝ => |p.1 - p.2| * u p.1 * u p.2) (Irest.prod Irest))
    (hVV : Integrable (fun p : ℝ × ℝ => |p.1 - p.2| * v p.1 * v p.2) (Irest.prod Irest))
    (hVW : Integrable (fun p : ℝ × ℝ => |p.1 - p.2| * v p.1 * (u p.2 - v p.2))
      (Irest.prod Irest))
    (hWV : Integrable (fun p : ℝ × ℝ => |p.1 - p.2| * (u p.1 - v p.1) * v p.2)
      (Irest.prod Irest))
    (hWW : Integrable (fun p : ℝ × ℝ => |p.1 - p.2| * (u p.1 - v p.1) * (u p.2 - v p.2))
      (Irest.prod Irest)) :
    Bfun u u - Bfun v v
      = Bfun v (fun t => u t - v t) + Bfun (fun t => u t - v t) v
        + Bfun (fun t => u t - v t) (fun t => u t - v t) := by
  have hcongr : ∫ p, (|p.1 - p.2| * u p.1 * u p.2 - |p.1 - p.2| * v p.1 * v p.2)
        ∂(Irest.prod Irest)
      = ∫ p, (|p.1 - p.2| * v p.1 * (u p.2 - v p.2)
          + |p.1 - p.2| * (u p.1 - v p.1) * v p.2
          + |p.1 - p.2| * (u p.1 - v p.1) * (u p.2 - v p.2)) ∂(Irest.prod Irest) :=
    MeasureTheory.integral_congr_ae (Filter.Eventually.of_forall (fun p => by ring))
  unfold Bfun
  rw [← MeasureTheory.integral_sub hUU hVV, hcongr]
  rw [MeasureTheory.integral_add (hVW.add hWV) hWW, MeasureTheory.integral_add hVW hWV]
